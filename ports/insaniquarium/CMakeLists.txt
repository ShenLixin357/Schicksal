cmake_minimum_required(VERSION 3.26)
project(SchicksalInsaniquariumWeb LANGUAGES C CXX)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Build this target with emcmake (Emscripten 4.0.20).")
endif()
if(NOT INSANIQUARIUM_SOURCE)
  message(FATAL_ERROR "Set INSANIQUARIUM_SOURCE to the prepared, pinned upstream checkout.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fexceptions -Wno-narrowing")
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "" FORCE)
set(POPLIB_WITH_BASS OFF CACHE BOOL "" FORCE)
set(SDLTTF_VENDORED ON CACHE BOOL "" FORCE)
set(SDLTTF_PLUTOSVG OFF CACHE BOOL "" FORCE)
set(SDLTTF_HARFBUZZ OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(MINIAUDIO_NO_DEVICEIO ON CACHE BOOL "" FORCE)
set(MINIAUDIO_NO_THREADING ON CACHE BOOL "" FORCE)
set(MINIAUDIO_NO_EXTRA_NODES ON CACHE BOOL "" FORCE)
add_subdirectory("${INSANIQUARIUM_SOURCE}/port" native)
set_target_properties(WinFish PROPERTIES
  OUTPUT_NAME insaniquarium RUNTIME_OUTPUT_NAME insaniquarium SUFFIX .mjs
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/engine"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/engine"
  RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/engine")
target_link_options(WinFish PRIVATE
  -sASYNCIFY=1 -sASYNCIFY_STACK_SIZE=262144
  -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=67108864 -sMAXIMUM_MEMORY=536870912
  -sSTACK_SIZE=4194304 -sMODULARIZE=1 -sEXPORT_ES6=1
  -sEXPORT_NAME=createInsaniquarium -sENVIRONMENT=web,node
  -sFORCE_FILESYSTEM=1 -sEXIT_RUNTIME=0 -sDISABLE_EXCEPTION_CATCHING=0
  "-sEXPORTED_FUNCTIONS=['_main','_schicksal_request_save']"
  "-sEXPORTED_RUNTIME_METHODS=['FS','callMain']"
  -lopenal)
